================================================================
  13. COMMON TROUBLESHOOTING (Windows)
================================================================

# Error: "streamlit not found"
#   Solution: Activate venv first
.\.venv\Scripts\Activate.ps1

# Error: "No API key found"
#   Solution: Add key to .env or use mock mode
notepad .env   # set GROQ_API_KEY or USE_MOCK_LLM=true

# Error: "No questions found for slot"
#   Solution: Import the question bank first (see Section 4)

# Error: "Cannot export: set is GENERATED"
#   Solution: In Faculty Review -> Approve all questions -> Approve Set -> Export

# Error: "Port 8501 already in use"
Get-Process -Name python -ErrorAction SilentlyContinue | Stop-Process -Force
streamlit run app.py

# Error: "Execution of scripts is disabled on this system"
#   Solution: Run this once in your PowerShell session:
Set-ExecutionPolicy -Scope Process -ExecutionPolicy Bypass
